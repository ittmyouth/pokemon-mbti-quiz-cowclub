let currentQuestionIndex = 0;
let scores = { E: 0, I: 0, S: 0, N: 0, T: 0, F: 0, J: 0, P: 0 };

function startQuiz() {
    document.getElementById('start-view').classList.add('hidden');
    document.getElementById('quiz-view').classList.remove('hidden');
    showQuestion();
}

function showQuestion() {
    if (currentQuestionIndex < questions.length) {
        let qData = questions[currentQuestionIndex];
        document.getElementById('progress').style.width = `${(currentQuestionIndex / questions.length) * 100}%`;
        document.getElementById('progress-text').innerText = `${currentQuestionIndex}/12`;
        document.getElementById('question').innerText = qData.q;
        document.getElementById('optA').innerText = qData.a;
        document.getElementById('optB').innerText = qData.b;
    } else {
        calculateResult();
    }
}

function selectOption(choice) {
    let qData = questions[currentQuestionIndex];
    if (choice === 'A') {
        scores[qData.valA]++;
    } else {
        scores[qData.valB]++;
    }
    currentQuestionIndex++;
    showQuestion();
}

function calculateResult() {
    document.getElementById('quiz-view').classList.add('hidden');
    document.getElementById('result-view').classList.remove('hidden');
    
    let mbti = "";
    mbti += (scores.E >= scores.I) ? "E" : "I";
    mbti += (scores.S >= scores.N) ? "S" : "N";
    mbti += (scores.T >= scores.F) ? "T" : "F";
    mbti += (scores.J >= scores.P) ? "J" : "P";
    
    let result = pokemonDatabase[mbti];
    
    /* 🌲 核心修改：直接讀取你本地 images 資料夾中對應小寫 mbti 名字的圖片（例：images/infp.png） */
    document.getElementById('pokemon-pic').src = `images/${mbti.toLowerCase()}.png`;
    document.getElementById('pokemon-name').innerText = result.name;
    
    let typeBadge = document.getElementById('pokemon-type');
    typeBadge.innerText = result.type;
    typeBadge.style.backgroundColor = result.color;

    document.getElementById('mbti-type').innerText = `【 靈魂型態：${mbti} 】`;
    document.getElementById('pokemon-desc').innerText = result.desc;
}

function restartQuiz() {
    currentQuestionIndex = 0;
    scores = { E: 0, I: 0, S: 0, N: 0, T: 0, F: 0, J: 0, P: 0 };
    document.getElementById('result-view').classList.add('hidden');
    document.getElementById('start-view').classList.remove('hidden');
}
